\newcommand{\declareGenericFig}[5]{
  \expandafter\newcommand\csname myFloat@#1\endcsname[2]{
    \begin{#4}[##1]
      \centering
      \includegraphics[width=#5\dimexpr ##2\linewidth \relax]{figures/#1}
      \caption[#2]{#2. #3}
      \label{fig:#1}
    \end{#4}
  }
}
\newcommand{\declareFig}[3]{\declareGenericFig{#1}{#2}{#3}{figure}{0.6}}
\newcommand{\declareWideFig}[3]{\declareGenericFig{#1}{#2}{#3}{figure}{1}}

\definecolor{myRoundedFigBorderColor}{HTML}{666666}
\newcommand{\declareRoundedGenericFig}[6]{
  \expandafter\newcommand\csname myFloat@#1\endcsname[2]{
    \begin{#5}[##1]
      \centering
      \begin{tikzpicture}
        \node[
          draw=myRoundedFigBorderColor,
          line width=1pt,
          rounded corners=5pt,
          inner sep=#2
        ] {\includegraphics[width=#6\dimexpr ##2\linewidth \relax]{figures/#1}};
      \end{tikzpicture}
      \caption[#3]{#3. #4}
      \label{fig:#1}
    \end{#5}
  }
}
\newcommand{\declareRoundedFig}[4]{\declareRoundedGenericFig{#1}{#2}{#3}{#4}{figure}{0.6}}
\newcommand{\declareRoundedWideFig}[4]{\declareRoundedGenericFig{#1}{#2}{#3}{#4}{figure}{1}}

\newcommand{\declareGenericTab}[6]{
  \expandafter\newcommand\csname myFloat@#1\endcsname[2]{
    \begin{#5}[##1]
      \centering
      \caption[#2]{#2. #3}
      \label{tab:#1}
      \resizebox{#6\dimexpr ##2\linewidth \relax}{!}{#4}
    \end{#5}
  }
}
\newcommand{\declareTab}[4]{\declareGenericTab{#1}{#2}{#3}{#4}{table}{0.6}}
\newcommand{\declareWideTab}[4]{\declareGenericTab{#1}{#2}{#3}{#4}{table}{1}}

\newcommand{\myFloat}[3]{\csname myFloat@#1\endcsname{#2}{#3}}
